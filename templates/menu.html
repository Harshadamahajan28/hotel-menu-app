let cart = []; function addToCart(itemName, price) { const taste = document.getElementById(`taste-${itemName}`) ? document.getElementById(`taste-${itemName}`).value : 'Medium'; // कार्टमध्ये आधीच हाच पदार्थ आणि तीच टेस्ट असेल तर Quantity वाढवणे const existingIndex = cart.findIndex(item => item.name === itemName && item.taste === taste); if (existingIndex > -1) { cart[existingIndex].qty += 1; } else { cart.push({ name: itemName, price: price, qty: 1, taste: taste }); } renderCart(); } // कार्टमधून आयटम Delete / Remove करण्याचे फंक्शन function removeFromCart(index) { cart.splice(index, 1); // Delete item renderCart(); } function renderCart() { const cartContainer = document.getElementById('cartItems'); let total = 0; cartContainer.innerHTML = ''; cart.forEach((item, index) => { total += item.price * item.qty; cartContainer.innerHTML += `
${item.name} (${item.taste})
₹${item.price} x ${item.qty}
`; }); document.getElementById('cartTotal').innerText = total; }
